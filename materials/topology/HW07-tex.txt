\begin{problem}
\begin{enumerate}
	\item Prove that any set endowed with the cofinite topology is compact.
	\item When is a set endowed with the cocountable topology compact? 
\end{enumerate}
\end{problem}

\begin{problem}
Let $X$ and $Y$ be subspaces of $\R^n$. For each of the following situations, determine whether $X$ and $Y$ can be homeomorphic
\begin{enumerate}
	\item $X$ closed and bounded, $Y$ closed but not bounded.
	\item $X$ bounded, $Y$ non-bounded.
	\item $X$ closed and bounded, $Y$ bounded but not closed.
	\item $X$ closed, $Y$ not closed.
\end{enumerate}
If you think that such $X$ and $Y$ exist, give an example (and justify). Otherwise
prove why no such pairs exist.
\end{problem}


\begin{problem} Using connectedness and compactness, prove that the following spaces are nor  homeomorphic 
\[ [0, 1], \quad [0,1), \quad  \R, \quad  \R^2, \quad  S^1.\]
\end{problem}
