\begin{problem}
Prove that composition of continuous functions is continuous. 
\end{problem}

\begin{problem}
Let $f\colon X \to Y$ be a function between topological spaces. Prove that the following are equivalent
\begin{enumerate}
	\item $f$ is continuous
	\item $f^{-1}(C)\subseteq X$ is closed for every closed set $C\subseteq Y$.
	\item If $\mathcal S$ is a subbasis for the topology of $Y$, then $f^{-1}(S)\subseteq X$ is open for every $S \in \mathcal S$.
\end{enumerate}
\end{problem}


\begin{problem}
Let $\{(X_i, \mathcal T_i) \mid i \in \Lambda\}$ be a collection of topological spaces and let $(X,\mathcal T)$ be a topological space. Prove that a function $f\colon (X,\mathcal T) \to \left(\prod_{i \in \Lambda} X_i, \mathcal{T}_{\text{prod}}\right)$ is continuous if and only if, for all $j\in\Lambda$, the coordinate function $f_j\colon (X,\mathcal T) \to (X_j,\mathcal T_j)$ is continuous.\footnote{This means that the product topology is better than the box topology from a categorical perspective.} 
\end{problem}

\begin{problem}
We say that a function $f\colon X \to Y$ between topological spaces is \emph{open} if $f(U)\subseteq Y$ is open for every open $U\subseteq X$. 
\begin{enumerate}
	\item Assume that $\mathcal B$ is a basis for the topology of $X$. Prove that $f$ is open if and only if $f(B)\subseteq Y$ is open for every $B\in\mathcal B$.
	\item Assume that $f$ is bijective. Prove that $f$ is open if and only if $f^{-1}\colon Y \to X$ is continuous. 
\end{enumerate}
\end{problem}

\begin{problem}
Let $f\colon X \to Y$ be a continuous function between topological spaces. Let $A\subseteq X$ and $B\subseteq Y$ be subspaces. Prove \textbf{two} of the following. 
\begin{enumerate}
	\item The induced topology on $A$ is the coarsest topology on $A$ such that the inclusion $\iota_A\colon A \to X$ is continuous. 
	\item The restriction $f\vert_A \colon A \to Y$ is continuous. 
	\item If $f(X)\subseteq B$, then corestriction $f\vert^B \colon X \to B$ is continuous.
\end{enumerate}
\end{problem}

\begin{problem}
Let $f\colon X \to Y$ be a function between topological spaces. Assume that $\{U_i\mid i \in \Lambda\}$ is a collection of open sets in $X$ such that $X=\bigcup_{i\in \Lambda} U_i$. Prove that $f$ is continuous if and only if, for all $i\in \Lambda$, the restriction $f \vert_{U_i} \colon U_i \to Y$ is continuous. 
\end{problem}
